Wait for the pods and the load balancer:
  kubectl -n {{ .Release.Namespace }} rollout status deploy/{{ include "lab06-app.fullname" . }}
  kubectl -n {{ .Release.Namespace }} get svc {{ include "lab06-app.fullname" . }} -w     # EXTERNAL-IP appears in ~2 min

Then:
  NLB=$(kubectl -n {{ .Release.Namespace }} get svc {{ include "lab06-app.fullname" . }} -o jsonpath='{.status.loadBalancer.ingress[0].hostname}')
  curl http://$NLB/healthz      # {"status":"ok"}  (DNS for a new NLB can take another minute)
  curl http://$NLB/version      # which build is live
  curl http://$NLB/metrics      # what Prometheus scrapes
  curl -i http://$NLB/boom      # 500 on purpose: feeds the 5xx alert
